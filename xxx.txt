    let clickAudioContext = null;
    let uiAudioCtx = null;

    function playClickSound() {
        try {
            const AudioCtx = window.AudioContext || window.webkitAudioContext;
            if (!AudioCtx) return;
            if (!clickAudioContext) clickAudioContext = new AudioCtx();
            if (clickAudioContext.state === 'suspended') clickAudioContext.resume();
            const now = clickAudioContext.currentTime;
            const osc = clickAudioContext.createOscillator();
            const gain = clickAudioContext.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(520, now);
            osc.frequency.exponentialRampToValueAtTime(760, now + 0.07);
            gain.gain.setValueAtTime(0.0001, now);
            gain.gain.exponentialRampToValueAtTime(0.12, now + 0.008);
            gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.09);
            osc.connect(gain);
            gain.connect(clickAudioContext.destination);
            osc.start(now);
            osc.stop(now + 0.1);
        } catch (e) {
            console.log('Dźwięk kliknięcia niedostępny:', e);
        }
    }

    function initUiAudio() {
        if (!uiAudioCtx) {
            uiAudioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        if (uiAudioCtx.state === 'suspended') {
            uiAudioCtx.resume();
        }
    }

    function playShuffleSound() {
        initUiAudio();
        const osc = uiAudioCtx.createOscillator();
        const gain = uiAudioCtx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(400, uiAudioCtx.currentTime);

    const canvas = document.getElementById('fxCanvas');
    const ctx = canvas.getContext('2d');
    resizeCanvas();

    function resizeCanvas() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;

    let particles = [];
    let musicNotes = [];
    let spotlightAngle = 0;
    let effectTimer = null;
    let fadeOutInterval = null;

    function createFirework(x, y) {
        const colors = ['#ffb703', '#2ecc71', '#00d4ff', '#e74c3c', '#9b59b6', '#ffffff'];
        for (let i = 0; i < 45; i++) {
            const angle = Math.random() * Math.PI * 2;
            const speed = Math.random() * 7 + 2;
            particles.push({
                x: x, y: y,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                alpha: 1,
                color: colors[Math.floor(Math.random() * colors.length)],
                size: Math.random() * 3 + 2
            });
        }
    }

    function createMusicNote() {
        const notesSymbols = ['♪', '♫', '♬', '♩'];
        musicNotes.push({
            x: Math.random() * canvas.width,
            y: canvas.height + 40,
            vy: Math.random() * 3 + 2,
            vx: (Math.random() - 0.5) * 2.5,
            alpha: 1,
            symbol: notesSymbols[Math.floor(Math.random() * notesSymbols.length)],
            size: Math.random() * 14 + 22
        });
    }

    function animateStudio() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        spotlightAngle += 0.015;
        ctx.save();
        ctx.globalAlpha = 0.12;

        let grad1 = ctx.createLinearGradient(0, 0, canvas.width / 2 + Math.sin(spotlightAngle) * 350, canvas.height);
        grad1.addColorStop(0, '#00d4ff'); grad1.addColorStop(1, 'transparent');
        ctx.fillStyle = grad1;
        ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(canvas.width / 2 + Math.sin(spotlightAngle) * 450, canvas.height); ctx.lineTo(0, canvas.height); ctx.fill();

        let grad2 = ctx.createLinearGradient(canvas.width, 0, canvas.width / 2 - Math.sin(spotlightAngle) * 350, canvas.height);
        grad2.addColorStop(0, '#ffb703'); grad2.addColorStop(1, 'transparent');
        ctx.fillStyle = grad2;
        ctx.beginPath(); ctx.moveTo(canvas.width, 0); ctx.lineTo(canvas.width / 2 - Math.sin(spotlightAngle) * 450, canvas.height); ctx.lineTo(canvas.width, canvas.height); ctx.fill();
        ctx.restore();

        musicNotes.forEach((n, index) => {
            n.y -= n.vy;
            n.x += n.vx;
            if (n.y < canvas.height * 0.2) n.alpha -= 0.02;

            ctx.save();
            ctx.globalAlpha = Math.max(n.alpha, 0);
            ctx.fillStyle = '#ffd700';
            ctx.font = `bold ${n.size}px Arial, sans-serif`;
            ctx.shadowColor = '#ffb703'; ctx.shadowBlur = 15;
            ctx.fillText(n.symbol, n.x, n.y);
            ctx.restore();

            if (n.y < -50 || n.alpha <= 0) musicNotes.splice(index, 1);
        });

        particles.forEach((p, index) => {
            p.x += p.vx; p.y += p.vy;
            p.vy += 0.12;
            p.alpha -= 0.018;

            ctx.save();
            ctx.globalAlpha = Math.max(p.alpha, 0);
            ctx.fillStyle = p.color;
            ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2); ctx.fill();
            ctx.restore();

            if (p.alpha <= 0) particles.splice(index, 1);
        });

        requestAnimationFrame(animateStudio);
    }
    animateStudio();

    function startStudioShowEffects() {
        effectTimer = setInterval(() => {
            const rx = Math.random() * (canvas.width - 150) + 75;
            const ry = Math.random() * (canvas.height * 0.55);
            createFirework(rx, ry);
            createMusicNote();
            createMusicNote();
        }, 280);
    }

    function stopStudioShowEffects() {
        if (effectTimer) {
            clearInterval(effectTimer);
            effectTimer = null;
        }
        if (fadeOutInterval) return;

        fadeOutInterval = setInterval(() => {
            let stillAlive = false;
            particles.forEach(p => { p.alpha -= 0.02; if(p.alpha > 0) stillAlive = true; });
            musicNotes.forEach(n => { n.alpha -= 0.02; if(n.alpha > 0) stillAlive = true; });
            
            if (!stillAlive) {
                clearInterval(fadeOutInterval);
                fadeOutInterval = null;
                particles = [];
                musicNotes = [];
            }
        }, 30);
    }

    function typeWriterEffect(text, elementId, speed = 45) {
        const el = document.getElementById(elementId);
        if(!el) return;
        el.innerHTML = "";
        const wrapper = document.getElementById('speechBubbleWrapper');
        if (wrapper) wrapper.classList.add('visible');
        let i = 0;

        function typing() {
            if (i < text.length) {
                el.innerHTML += text.charAt(i);
                i++;
                setTimeout(typing, speed);
            }
        }
        typing();
    }

    function getReviewSongs() {
        try {
            let saved = localStorage.getItem('kindziu_review_songs_v2');
            if (!saved) return [];
            return JSON.parse(saved);
        } catch (e) {
            return [];
        }
    }

    function addReviewSong(song) {
        try {
            let list = getReviewSongs();
            if (!list.some(s => s.query === song.query)) {
                list.push({
                    category: song.category,
                    query: song.query,
                    title: song.title
                });
                localStorage.setItem('kindziu_review_songs_v2', JSON.stringify(list));
            }
            updateReviewMenuButton();
        } catch (e) {
            console.error("Błąd zapisu do powtórek:", e);
        }
    }

    function removeReviewSong(query) {
        try {
            let list = getReviewSongs();
            let filtered = list.filter(s => s.query !== query);
            localStorage.setItem('kindziu_review_songs_v2', JSON.stringify(filtered));
            updateReviewMenuButton();
        } catch (e) {
            console.error("Błąd usuwania z powtórek:", e);
        }
    }

    function playFeedbackSound(isWin) {
        try {
            const AudioCtx = window.AudioContext || window.webkitAudioContext;
            const ctx = new AudioCtx();
            if (ctx.state === 'suspended') ctx.resume();
            const now = ctx.currentTime;
            const notes = isWin ? [523.25, 659.25, 783.99] : [349.23, 311.13];
            notes.forEach((f, i) => {
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(f, now + i * 0.08);
                gain.gain.setValueAtTime(0.0001, now + i * 0.08);
                gain.gain.exponentialRampToValueAtTime(0.15, now + i * 0.08 + 0.02);
                gain.gain.exponentialRampToValueAtTime(0.0001, now + i * 0.08 + 0.3);
                osc.connect(gain); gain.connect(ctx.destination);
                osc.start(now + i * 0.08); osc.stop(now + i * 0.08 + 0.35);
            });
        } catch (e) {
            console.log("Audio niedostępne", e);
        }
    }

    function drawNewSet() {
        if (isReviewMode) {
            startReviewMode();
            return;
        }

        playShuffleSound();
        const icon = document.getElementById('reloadIcon');
        if (icon) {
            icon.classList.remove('spin-anim');
            void icon.offsetWidth;
            icon.classList.add('spin-anim');
        }

        playedIds = [];
        const requiredCount = currentRound === 1 ? 5 : 3;
        
        const categoriesList = [...new Set(masterSongPool.map(item => item.category))];
        const shuffledCategories = categoriesList.sort(() => 0.5 - Math.random());
        const selectedCategories = shuffledCategories.slice(0, requiredCount);

        activeCategoriesSet = selectedCategories.map((catName, index) => {
            const availableSongs = masterSongPool.filter(s => s.category === catName);
            const randomSong = availableSongs[Math.floor(Math.random() * availableSongs.length)];
            return {
                id: index + 1,
                category: catName,
                query: randomSong.query,
                title: randomSong.title
            };
        });

        renderCategories();
    }

    function renderCategories() {
        const list = document.getElementById('categoriesList');
        list.innerHTML = '';

        activeCategoriesSet.forEach((item, index) => {
            const btn = document.createElement('button');
            let isPlayed = playedIds.includes(item.id);
            let delay = 0;
            
            if (playedIds.length === 0) {
                btn.className = 'cat-btn animate-in';
                delay = index * 0.12;
                btn.style.animationDelay = `${delay}s`;
                playPopSound(delay + 0.05);
            } else {
                btn.className = 'cat-btn' + (isPlayed ? ' played' : '');
                btn.style.opacity = '1';
                btn.style.transform = 'translateY(0)';
            }

            btn.innerHTML = `<span>${item.category}</span> <span>${isPlayed ? '✓' : '➔'}</span>`;
            if (!isPlayed) {
                btn.onclick = () => startSong(item);
            }
            list.appendChild(btn);
        });
    }

    async function startSong(song) {
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel();
        }
        selectedSong = song;
        document.getElementById('categoryScreen').style.display = 'none';
        document.getElementById('gameScreen').style.display = 'block';
        document.getElementById('currentCategory').innerText = song.category;
        homeBtn.style.display = 'flex';
        
        answerBox.style.display = 'none';
        revealBtn.style.display = 'none';
        stopBtn.disabled = true;
        statusMsg.innerText = "Szukanie utworu w chmurze...";
        progressBar.style.width = "0%";

        elapsedSeconds = 0;
        finalAmount = (isReviewMode || currentRound === 1) ? 100 : 200;
        moneyDisplay.innerText = finalAmount;

        try {
            const cleanQuery = song.query.replace(/[^\w\s]/gi, '');
            const targetUrl = `https://itunes.apple.com/search?term=${encodeURIComponent(cleanQuery)}&entity=song&limit=1`;
            
            const response = await fetch(targetUrl);
            const data = await response.json();

            if (data.results && data.results.length > 0 && data.results[0].previewUrl) {
                currentPreviewUrl = data.results[0].previewUrl;
                audio.src = currentPreviewUrl;
                audio.currentTime = 12; 
                
                audio.play();
                statusMsg.innerText = "Graj!";
                stopBtn.disabled = false;

                clearInterval(timerInterval);
                timerInterval = setInterval(() => {
                    elapsedSeconds++;
                    finalAmount = (isReviewMode || currentRound === 1) ? (100 + elapsedSeconds * 10) : (200 - elapsedSeconds * 10);
                    moneyDisplay.innerText = finalAmount;
                    
                    let progressPercent = (elapsedSeconds / 10) * 100;
                    progressBar.style.width = Math.min(progressPercent, 100) + "%";

                    if (elapsedSeconds >= 10) stopMusic(true);
                }, 1000);
            } else {
                statusMsg.innerText = "Błąd: Brak darmowej próbki w bazie iTunes dla tego utworu.";
            }
        } catch (e) {
            statusMsg.innerText = "Błąd połączenia z serwerem muzycznym.";
        }
    }

        do {
            randomIndex = Math.floor(Math.random() * stopReactionSounds.length);
        } while (recentReactionIndices.includes(randomIndex));

        recentReactionIndices.push(randomIndex);
        if (recentReactionIndices.length > 3) recentReactionIndices.shift();

        stopReactionAudio.src = stopReactionSounds[randomIndex];
        stopReactionAudio.currentTime = 0;
        stopReactionAudio.play().catch(e => console.log('Błąd audio reakcji:', e));
    }

    function stopMusic(timeOut = false) {
        audio.pause();
        clearInterval(timerInterval);
        stopBtn.disabled = true;
        revealBtn.style.display = 'block';

        if (timeOut) {
            statusMsg.innerText = "Czas minął!";
            document.getElementById('resultText').innerText = "Brak reakcji w 10 sekund. Zdobyto: 0 zł";
            finalAmount = 0;
        } else {
            statusMsg.innerText = "Zatrzymano!";
            document.getElementById('resultText').innerText = "Do Twojego konta trafia: " + finalAmount + " zł";
            totalSessionMoney += finalAmount;

            setTimeout(() => {
                playRandomStopReaction();
            }, 500);
        }
    }

    function showSplitButtons() {
        document.getElementById('mainSpeakerBtn').style.display = 'none';
        document.getElementById('splitSpeakerContainer').style.display = 'flex';
    }

    function speakWithLang(langCode) {
        if (!selectedSong) return;
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel();
            const utterance = new SpeechSynthesisUtterance(selectedSong.title);
            utterance.lang = langCode;
            utterance.rate = 0.88;
            window.speechSynthesis.speak(utterance);
        } else {
            alert("Twoja przeglądarka nie obsługuje syntezatora mowy.");
        }
    }

    function revealAnswer() {
        document.getElementById('answerText').innerText = selectedSong.title;
        
        document.getElementById('mainSpeakerBtn').style.display = 'block';
        document.getElementById('splitSpeakerContainer').style.display = 'none';

        let labelYes = isReviewMode ? "Już znam" : "Znałam";
        let labelNo = isReviewMode ? "Nie znałam" : "Nie znałam";

        let feedbackHtml = `
            <button class="feedback-btn" onclick="handleFeedback(true)" style="background: #2ecc71;">✅ ${labelYes}</button>
            <button class="feedback-btn" onclick="handleFeedback(false)" style="background: #e74c3c;">❌ ${labelNo}</button>
        `;
        
        let existingFeedback = document.getElementById('feedbackBtns');
        existingFeedback.innerHTML = feedbackHtml;
        existingFeedback.style.display = 'flex';

        answerBox.style.display = 'block';
        revealBtn.style.display = 'none';
        if (!playedIds.includes(selectedSong.id)) playedIds.push(selectedSong.id);
    }

    let isFullPlaying = false;
    function toggleFullPreview() {
        const btn = document.getElementById('listenFullBtn');
        if (!currentPreviewUrl) return;

        if (!isFullPlaying) {
            audio.src = currentPreviewUrl;
            audio.currentTime = 0;
            audio.play().then(() => {
                isFullPlaying = true;
                btn.innerText = "⏹ Zatrzymaj odsłuch";
                btn.style.background = "linear-gradient(180deg, #e74c3c 0%, #c0392b 100%)";
            }).catch(e => console.log('Błąd odsłuchu:', e));
        } else {
            audio.pause();
            audio.currentTime = 0;
            isFullPlaying = false;
            btn.innerText = "🎧 Posłuchaj całości";
            btn.style.background = "linear-gradient(180deg, #3a86ff 0%, #2563eb 100%)";
        }
    }

    function handleFeedback(isKnown) {
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel();
        }
        if (isFullPlaying) {
            audio.pause();
            isFullPlaying = false;
            const btn = document.getElementById('listenFullBtn');
            if(btn) {
                btn.innerText = "🎧 Posłuchaj całości";
                btn.style.background = "linear-gradient(180deg, #3a86ff 0%, #2563eb 100%)";
            }
        }

        playFeedbackSound(isKnown);

        let fb = document.getElementById('feedbackBtns');
        if (fb) fb.style.display = 'none';

        if (isReviewMode) {
            if (isKnown) {
                removeReviewSong(selectedSong.query);
            }
        } else {
            if (!isKnown) {
                addReviewSong(selectedSong);
            }
        }
    }

    function backToCategories() {
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel();
        }
        if (isFullPlaying) {
            audio.pause();
            isFullPlaying = false;
        }

        const stopReactionAudio = document.getElementById('stopReactionAudio');
        if (stopReactionAudio) {
            stopReactionAudio.pause();
            stopReactionAudio.currentTime = 0;
        }

        document.getElementById('gameScreen').style.display = 'none';
        
        if (playedIds.length >= activeCategoriesSet.length) {
            showSummaryScreen();
        } else {
            document.getElementById('categoryScreen').style.display = 'block';
            renderCategories();
        }
    }

    function showSummaryScreen() {
        document.getElementById('summaryScreen').style.display = 'block';
        document.getElementById('totalMoneyDisplay').innerText = totalSessionMoney;
        homeBtn.style.display = 'flex';
        updateReviewMenuButton();
    }

        updateReviewMenuButton();
    };
